"use client";

import React, { useEffect } from "react";
import gsap from "gsap";
import ScrollTrigger from "gsap/ScrollTrigger";
import Lenis from "@studio-freight/lenis";

gsap.registerPlugin(ScrollTrigger);

export default function Page() {
  useEffect(() => {
    /* ================= LENIS ================= */
    const lenis = new Lenis({
      duration: 1.2,
      easing: (t) => Math.min(1, 1.001 - Math.pow(2, -10 * t)),
      smooth: true,
      smoothTouch: true,
      touchMultiplier: 2,
    });

    function raf(time) {
      lenis.raf(time);
      ScrollTrigger.update();
      requestAnimationFrame(raf);
    }
    requestAnimationFrame(raf);

    /* ================= Z-INDEX ================= */
    document.querySelectorAll(".img-wrapper").forEach((el) => {
      const order = el.getAttribute("data-index");
      if (order) el.style.zIndex = order;
    });

    /* ================= GSAP ================= */
    const imgs = gsap.utils.toArray(".img-wrapper img");
    const bgColors = ["#362F4F", "#5B23FF", "#008BFF", "#E4FF30"];

    ScrollTrigger.matchMedia({
      "(min-width: 769px)": () => {
        const mainTimeline = gsap.timeline({
          scrollTrigger: {
            trigger: ".arch",
            start: "top top",
            end: "bottom bottom",
            pin: ".arch__right",
            scrub: true,
          },
        });

        gsap.set(imgs, {
          clipPath: "inset(0)",
          objectPosition: "0px 0%",
        });

        imgs.forEach((_, index) => {
          const currentImage = imgs[index];
          const nextImage = imgs[index + 1];
          if (!nextImage) return;

          const tl = gsap.timeline();

          /* =================
             SCROLL 1
             Text + Video
          ================= */
          tl.to({}, { duration: 1 });

          /* =================
             SCROLL 2
             Left Grid
          ================= */
          tl.to({}, { duration: 1 });

          /* =================
             SCROLL 3
             RIGHT IMAGE + BG
             (UNCHANGED LOGIC)
          ================= */
          tl.to(
            ".full-width-container",
            {
              backgroundColor: bgColors[index],
              duration: 1.5,
              ease: "power2.inOut",
            },
            0
          )
            .to(
              currentImage,
              {
                clipPath: "inset(0px 0px 100%)",
                objectPosition: "0px 60%",
                duration: 1.5,
                ease: "none",
              },
              0
            )
            .to(
              nextImage,
              {
                objectPosition: "0px 40%",
                duration: 1.5,
                ease: "none",
              },
              0
            );

          mainTimeline.add(tl);
        });
      },
    });

    return () => {
      ScrollTrigger.killAll();
      lenis.destroy();
    };
  }, []);

  const events = [
    "Gusto Cultural Fest",
    "B.Parliament Debate",
    "Navratri Celebrations",
    "Sunidhi Live Concert",
  ];

  const images = [
    "/events/gusto.jpg",
    "/events/BParak.jpg",
    "/events/Navratri.JPG",
    "/events/Sunnidhi.jpg",
  ];

  return (
    <>
      {/* ================= GSAP SECTION ================= */}
      <div className="full-width-container">
        <div className="arch">
          {/* ================= LEFT ================= */}
          <div className="arch__left">
            {events.map((title, i) => (
              <React.Fragment key={i}>
                {/* TEXT + VIDEO */}
                <div className="arch__info">
                  <div className="content">
                    <h2 className="header">{title}</h2>
                    <p className="desc">
                      Replace this text with event-specific content.
                    </p>

                    <video
                      className="chanakya-video"
                      src="/chanu/chanudancev1.mp4"
                      autoPlay
                      loop
                      muted
                      playsInline
                    />
                  </div>
                </div>

                {/* GRID — SECOND SCROLL */}
                <div className="arch__grid">
                  <div className="arch__grid-full">
                    <img src={images[i]} alt="" />
                  </div>
                  <div className="arch__grid-row">
                    <img src={images[i]} alt="" />
                    <img src={images[i]} alt="" />
                  </div>
                </div>
              </React.Fragment>
            ))}
          </div>

          {/* ================= RIGHT (UNCHANGED) ================= */}
          <div className="arch__right">
            {images.map((src, i) => (
              <div className="img-wrapper" data-index={4 - i} key={i}>
                <img src={src} alt="" />
              </div>
            ))}
          </div>
        </div>
      </div>
    </>
  );
}
